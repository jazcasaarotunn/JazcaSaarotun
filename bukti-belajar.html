Harap isi semua kolom a, b, dan m dengan benar!`; return; } if (m <= 0) { hasilEl.innerHTML = `Error
Modulus m harus berupa bilangan bulat positif (m > 0)!`; return; } const selisih = a - b; const sisaMod = selisih % m; const isKongruen = (sisaMod === 0); if (isKongruen) { hasilEl.innerHTML = ` Status: KONGRUEN
Pernyataan: ${a} ≡ ${b} (mod ${m})
Proses: Selisih (a - b) = (${a} - ${b}) = ${selisih}. Karena ${m} membagi habis ${selisih} (${selisih} / ${m} = ${selisih / m}), maka pernyataan BENAR dan Kongruen. `; } else { hasilEl.innerHTML = ` Status: TIDAK KONGRUEN
Pernyataan: ${a} ≢ ${b} (mod ${m})
Proses: Selisih (a - b) = (${a} - ${b}) = ${selisih}. Karena ${m} tidak habis membagi ${selisih}, maka pernyataan SALAH. `; } } // LEVEL 5: Kalkulator Sistem Bilangan (Biner, Desimal, Oktal, Heksadesimal dengan Validasi Basis) function konversiSistemBilangan() { const inputVal = document.getElementById('numInput').value.trim(); const base = parseInt(document.getElementById('numBase').value); const binValEl = document.getElementById('binVal'); const decValEl = document.getElementById('decVal'); const octValEl = document.getElementById('octVal'); const hexValEl = document.getElementById('hexVal'); if (!inputVal) { alert('Harap masukkan bilangan yang ingin dikonversi!'); return; } // Validasi input sesuai basis bilangan yang dipilih let regex; if (base === 2) regex = /^[01]+$/; else if (base === 8) regex = /^[0-7]+$/; else if (base === 10) regex = /^[0-9]+$/; else if (base === 16) regex = /^[0-9a-fA-F]+$/; if (!regex.test(inputVal)) { alert(`Input "${inputVal}" tidak valid untuk basis ${base} yang dipilih! Harap masukkan karakter angka yang sesuai.`); return; } let decimalValue = parseInt(inputVal, base); if (isNaN(decimalValue)) { alert('Format bilangan tidak valid.'); return; } // Tampilkan hasil konversi ke empat basis if (binValEl) binValEl.textContent = decimalValue.toString(2); if (decValEl) decValEl.textContent = decimalValue.toString(10); if (octValEl) octValEl.textContent = decimalValue.toString(8); if (hexValEl) hexValEl.textContent = decimalValue.toString(16).toUpperCase(); } // LEVEL 6: Kalkulator Kombinatorika (Permutasi & Kombinasi) function hitungKombinatorika() { const n = parseInt(document.getElementById('combN').value); const r = parseInt(document.getElementById('combR').value); const hasilEl = document.getElementById('resCombinatorics'); if (isNaN(n) || isNaN(r) || n < 0 || r < 0) { hasilEl.innerHTML = `Error
Masukkan nilai n dan r berupa bilangan bulat non-negatif (>= 0).`; return; } if (r > n) { hasilEl.innerHTML = `Error
Nilai pilihan r tidak boleh lebih besar dari jumlah objek n!`; return; } function hitungFaktorial(num) { if (num === 0 || num === 1) return 1; let hasil = 1; for (let i = 2; i <= num; i++) { hasil *= i; } return hasil; } const factN = hitungFaktorial(n); const factNR = hitungFaktorial(n - r); const factR = hitungFaktorial(r); const permutasi = factN / factNR; const kombinasi = factN / (factR * factNR); hasilEl.innerHTML = ` Berhasil Dihitung
🧮 Permutasi P(${n},${r}): ${permutasi.toLocaleString('id-ID')}
📊 Kombinasi C(${n},${r}): ${kombinasi.toLocaleString('id-ID')}
`; }
